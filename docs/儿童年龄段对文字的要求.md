# 绘本创作工具 - 年龄段配置
# 用于控制生成的故事文本的复杂度、长度和风格

age_groups:
  # 婴幼儿 (1-3岁)
  - name: "婴幼儿 (1-3岁)"
    key: "toddler"
    # 文字密度：极低，每页1-2个短词或简单短语
    max_words_per_page: 8
    avg_words_per_page: 4
    # 句子长度（平均单词数）
    avg_sentence_length: 2
    # 总页数范围（最小-最大）
    page_range: [10, 16]
    # 重复句式：重要，便于模仿和记忆
    use_repetition: true
    # 词汇类型：拟声词、日常名词（妈妈、狗狗、球）、简单动词（抱抱、吃）
    vocabulary: "日常名词、拟声词、基础动词"
    # 概念复杂度：单一动作或情绪，无因果推理
    concept_complexity: "非常低"
    # 提示词模板 - 用于 LLM 生成符合要求的绘本文字
    prompt_template: |
      你是一位婴幼儿绘本作家。请为1-3岁幼儿创作一个故事。
      要求：
      - 每页文字不超过8个单词，平均4个单词。
      - 每个句子不超过3个单词。
      - 大量使用重复句式（如“抱抱，抱抱。”“小狗跑，小狗跳。”）。
      - 词汇仅包含：日常名词（妈妈、爸爸、猫、狗、球、奶瓶）、拟声词（汪汪、喵喵、嘭嘭）、简单动词（吃、睡、玩、抱）。
      - 总页数：{{page_count}}页。
      - 故事主题：{{story_topic}}（例如：找妈妈、睡觉时间、玩球）。
      请输出每一页的文字内容，用“页1：”“页2：”标注。

  # 低幼 (3-5岁)
  - name: "低幼 (3-5岁)"
    key: "preschool"
    max_words_per_page: 20
    avg_words_per_page: 12
    avg_sentence_length: 5
    page_range: [16, 24]
    use_repetition: true
    vocabulary: "常见名词、形容词（大/小/开心/难过）、简单时间词（今天/然后）"
    concept_complexity: "低，包含简单顺序事件（先...然后...），基本因果关系"
    prompt_template: |
      你是一位绘本作家，为3-5岁儿童创作故事。
      要求：
      - 每页文字10-20个单词，平均12个单词。
      - 句子长度4-6个单词，偶尔8个单词的长句。
      - 可以适度重复句式（例如每两页重复一次），但不必每页重复。
      - 词汇使用常见名词、基础形容词（大、小、开心、生气）和简单连接词（然后、因为）。
      - 故事有清晰的开始、中间和结尾，包含简单的问题和解决。
      - 总页数：{{page_count}}页。
      - 故事主题：{{story_topic}}。
      输出格式：页数及对应文字。

  # 学龄前/早期阅读 (5-7岁)
  - name: "早期阅读 (5-7岁)"
    key: "early_reader"
    max_words_per_page: 35
    avg_words_per_page: 25
    avg_sentence_length: 8
    page_range: [24, 32]
    use_repetition: false
    vocabulary: "丰富名词、常用动词变位、时间/地点状语、少量复合句连接词（虽然、但是）"
    concept_complexity: "中等，包含多步骤事件、人物动机、简单隐喻"
    prompt_template: |
      你为5-7岁儿童创作一个绘本故事，帮助从“听读”过渡到“自读”。
      要求：
      - 每页20-35个单词，平均25个单词。
      - 句子长度6-10个单词，允许少量12词的长句。
      - 避免过度重复，用不同的句式和词汇。
      - 包含对话（引导台词），有明确的人物和情节转折。
      - 总页数：{{page_count}}页。
      - 故事主题：{{story_topic}}。
      请逐页输出故事文本。

  # 大童 (7-9岁)
  - name: "大童 (7-9岁)"
    key: "older"
    max_words_per_page: 50
    avg_words_per_page: 40
    avg_sentence_length: 12
    page_range: [32, 40]
    use_repetition: false
    vocabulary: "较抽象词汇（勇气、友谊、信任）、复杂动词时态、复合句、习语"
    concept_complexity: "中高，包含多线程情节、角色冲突、道德困境"
    prompt_template: |
      你为7-9岁儿童创作章节绘本（文字量较大的绘本）。
      要求：
      - 每页35-50个单词，平均40个单词。
      - 句子长度8-15个单词，包含复合句和从句。
      - 使用丰富的描写和较抽象的词汇。
      - 故事可以有次要情节或反转，鼓励批判性思考。
      - 总页数：{{page_count}}页。
      - 故事主题：{{story_topic}}。
      请输出每页的完整文字。
